{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "open-terminal.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "Application: http://$NODE_IP:$NODE_PORT"
  echo "API docs: http://$NODE_IP:$NODE_PORT/docs"
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "open-terminal.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo "Application: http://$SERVICE_IP:{{ .Values.service.port }}"
  echo "API docs: http://$SERVICE_IP:{{ .Values.service.port }}/docs"
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "open-terminal.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:{{ .Values.openTerminal.server.port }} to use your application"
  echo "API documentation: http://127.0.0.1:{{ .Values.openTerminal.server.port }}/docs"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME {{ .Values.openTerminal.server.port }}:{{ .Values.openTerminal.server.port }}
